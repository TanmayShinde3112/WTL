<body>
  <h2>My Personalized Todo List</h2> <!-- Heading -->

  <input id="taskInput" placeholder="Add a task" /> <!-- Input for new task -->
  <button onclick="AddTask()">ADD TASK</button> <!-- Button calls AddTask function -->

  <ul id="tasklist"></ul> <!-- Empty unordered list for tasks -->

  <script>
    // Function to add a new task when called
    function AddTask() {
      let input = document.getElementById('taskInput'); // Get input element
      let text = input.value.trim(); // Get trimmed input value

      if (!text) { // If input empty
        return alert('Enter the task'); // Alert user and stop function
      }

      let li = document.createElement('li'); // Create new list item

      // Task text span
      let span = document.createElement('span'); // Create span to hold task text
      span.textContent = text; // Set span text to input value

      // Toggle strike-through on task text click
      span.onclick = () => {
        // If already line-through, remove it; else apply it
        span.style.textDecoration =
          span.style.textDecoration === 'line-through' ? '' : 'line-through';
      };

      // Edit button
      let editBtn = document.createElement('button'); // Create edit button
      editBtn.textContent = 'Edit'; // Button label
      editBtn.onclick = () => {
        let newText = prompt('Edit Text:', span.textContent); // Prompt for new text
        if (newText) {
          span.textContent = newText.trim(); // Update span text
          span.style.textDecoration = ''; // Remove strike-through after edit
        }
      };

      // Delete button
      let deleteBtn = document.createElement('button'); // Create delete button
      deleteBtn.textContent = 'Delete'; // Button label
      deleteBtn.onclick = () => {
        if (confirm('Delete Task?')) li.remove(); // Confirm and remove task
      };

      li.appendChild(span); // Add text span to the list item
      li.appendChild(editBtn); // Add edit button to the list item
      li.appendChild(deleteBtn); // Add delete button to the list item

      document.getElementById('tasklist').appendChild(li); // Add new task to the list

      input.value = ''; // Clear input box after adding
    }
  </script>
